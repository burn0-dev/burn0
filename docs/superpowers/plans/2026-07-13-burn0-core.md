# burn0 Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `@burn0/core` — zero-code HTTP capture, SQLite storage, LLM cost computation, and the query API the CLI will render.

**Architecture:** In-process interception (undici diagnostics_channel primary, fetch/http patches fallback) feeds an enrichment pipeline (entity detection → endpoint normalization → LLM usage parse → cost → redaction) into a batched write queue over better-sqlite3 (WAL). Readers use `query/` functions only. Every burn0 codepath is wrapped; failures never propagate to the host app.

**Tech Stack:** TypeScript (ESM+CJS via tsdown), better-sqlite3 behind a `StorageDriver` adapter, vitest, pnpm workspaces.

## Global Constraints

- Node >= 20.6 (`node --import` support). `"engines": { "node": ">=20.6" }`.
- DB path: `~/.burn0/burn0.db`; env `BURN0_DB_PATH` overrides (tests MUST set it to a temp dir).
- Retention defaults: payloads 7d (ceiling 14), metadata 30d (ceiling 90). Body cap 1 MiB (1048576 bytes).
- Disabled when: `BURN0_DISABLED=1`, or `NODE_ENV=production` without `BURN0_FORCE=1`, or non-Node runtime (bun/deno) → silent no-op (one stderr line for non-Node).
- Redaction ON by default. Redacted value is exactly `[redacted]`.
- No burn0 error may ever throw into the host app: every public entry wrapped in try/catch that increments an internal counter.
- Schema version 1. Breaking mismatch → rename db to `burn0.db.bak`, create fresh.
- Commit after every task (git repo already initialized in `burn0-1`).

```ts
import { HOSTS } from './hosts.js'

const TWO_LEVEL_TLDS = new Set(['co.uk', 'com.au', 'co.jp', 'co.in', 'com.br', 'co.nz'])

export function detectEntity(hostname: string): { entity: string; entityClass: 'llm' | 'general'; provider: string | null } {
  for (const rule of HOSTS) {
    if (rule.match(hostname)) return { entity: rule.entity, entityClass: rule.entityClass, provider: rule.provider }
  }
  return { entity: registrableDomain(hostname), entityClass: 'general', provider: null }
}

function registrableDomain(hostname: string): string {
  const parts = hostname.split('.')
  if (parts.length <= 2) return hostname
  const lastTwo = parts.slice(-2).join('.')
  const take = TWO_LEVEL_TLDS.has(lastTwo) ? 3 : 2
  return parts.slice(-take).join('.')
}

const NUMERIC = /^\d+$/
const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i
const LONG_HEX = /^[0-9a-f]{16,}$/i

export function normalizeEndpoint(pathname: string): string {
  const segments = pathname.split('/')
  const out = segments.map((seg) => {
    if (seg === '') return seg
    if (NUMERIC.test(seg) || UUID.test(seg) || LONG_HEX.test(seg)) return ':id'
    return seg
  })
  return out.join('') === '' ? '/' : out.join('/')
}
```

```ts
const SECRET_HEADERS = new Set(['authorization', 'x-api-key', 'api-key', 'x-goog-api-key', 'cookie', 'set-cookie', 'proxy-authorization'])
const SK_KEY = /sk-[A-Za-z0-9_-]{10,}/g
const BEARER = /(Bearer\s+)[A-Za-z0-9._~+/-]{10,}=*/g

export function redactHeaders(headers: Record<string, string>, extra: string[] = []): Record<string, string> {
  const extraSet = new Set(extra.map((h) => h.toLowerCase()))
  const out: Record<string, string> = {}
  for (const [k, v] of Object.entries(headers)) {
    const lk = k.toLowerCase()
    out[k] = SECRET_HEADERS.has(lk) || extraSet.has(lk) ? '[redacted]' : v
  }
  return out
}

export function redactBody(body: string | null): string | null {
  if (body == null) return body
  return body.replace(SK_KEY, '[redacted]').replace(BEARER, '$1[redacted]')
}
```

`packages/core/src/llm/parse.ts`:
```ts
export interface Usage {
  tokensIn: number | null
  tokensOut: number | null
  tokensCached: number | null
}

function nonNeg(n: unknown): number | null {
  return typeof n === 'number' && n >= 0 ? n : null
}

function usageFromObject(o: any): Usage | null {
  if (!o || typeof o !== 'object') return null
  if (o.usage && typeof o.usage === 'object') {
    const u = o.usage
    const tokensIn = nonNeg(u.prompt_tokens) ?? nonNeg(u.input_tokens)
    const tokensOut = nonNeg(u.completion_tokens) ?? nonNeg(u.output_tokens)
    const tokensCached = nonNeg(u.prompt_tokens_details?.cached_tokens) ?? nonNeg(u.cache_read_input_tokens)
    if (tokensIn !== null || tokensOut !== null) return { tokensIn, tokensOut, tokensCached }
  }
  if (o.usageMetadata && typeof o.usageMetadata === 'object') {
    const m = o.usageMetadata
    const tokensIn = nonNeg(m.promptTokenCount)
    const tokensOut = nonNeg(m.candidatesTokenCount)
    if (tokensIn !== null || tokensOut !== null) return { tokensIn, tokensOut, tokensCached: nonNeg(m.cachedContentTokenCount) }
  }
  return null
}

export function parseUsage(resBody: string | null): Usage | null {
  if (!resBody) return null
  try {
    return usageFromObject(JSON.parse(resBody))
  } catch {
    return null
  }
}

export function parseSSEUsage(raw: string): Usage | null {
  let tokensIn: number | null = null
  let tokensOut: number | null = null
  let tokensCached: number | null = null
  let found = false
  for (const line of raw.split('\n')) {
    if (!line.startsWith('data: ') || line === 'data: [DONE]') continue
    let parsed: any
    try { parsed = JSON.parse(line.slice(6)) } catch { continue }
    // Anthropic: usage split across message_start (input) and message_delta (output)
    const msgUsage = parsed?.message?.usage
    if (msgUsage) {
      tokensIn = nonNeg(msgUsage.input_tokens) ?? tokensIn
      tokensCached = nonNeg(msgUsage.cache_read_input_tokens) ?? tokensCached
      found = tokensIn !== null || found
    }
    if (parsed?.type === 'message_delta' && parsed.usage) {
      tokensOut = nonNeg(parsed.usage.output_tokens) ?? tokensOut
      found = true
    }
    // OpenAI-style: full usage object in a (usually final) chunk
    const u = usageFromObject(parsed)
    if (u) {
      tokensIn = u.tokensIn ?? tokensIn
      tokensOut = u.tokensOut ?? tokensOut
      tokensCached = u.tokensCached ?? tokensCached
      found = true
    }
  }
  return found ? { tokensIn, tokensOut, tokensCached } : null
}

export function extractModel(reqBody: string | null, urlPath: string, resBody: string | null): string | null {
  if (reqBody) {
    try {
      const m = JSON.parse(reqBody)?.model
      if (typeof m === 'string') return m
    } catch {}
  }
  const pathMatch = urlPath.match(/\/models\/([^/:?]+)/)
  if (pathMatch) return pathMatch[1]
  if (resBody) {
    try {
      const m = JSON.parse(resBody)?.model
      if (typeof m === 'string') return m
    } catch {}
  }
  return null
}
```

```ts
import { join, dirname } from 'node:path'
import { fileURLToPath } from 'node:url'
import type { Usage } from '../llm/parse.js'

export interface PriceEntry {
  input_cost_per_token?: number
  output_cost_per_token?: number
  cache_read_input_token_cost?: number
}
export type PricingTable = Record<string, PriceEntry>

export const PRICING_URL =
  'https://raw.githubusercontent.com/BerriAI/litellm/main/model_prices_and_context_window.json'

interface CacheFile { fetchedAt: number; table: PricingTable }

function readCache(cacheDir: string): CacheFile | null {
  try {
    return JSON.parse(readFileSync(join(cacheDir, 'pricing.json'), 'utf8')) as CacheFile
  } catch {
    return null
  }
}

function readSnapshot(): PricingTable {
  const here = dirname(fileURLToPath(import.meta.url))
  // works from src (tests) and from dist (snapshot.json copied at build)
  for (const p of [join(here, 'snapshot.json'), join(here, 'pricing', 'snapshot.json')]) {
    try { return JSON.parse(readFileSync(p, 'utf8')) } catch {}
  }
  return {}
}

export async function loadPricing(opts: {
  cacheDir: string
  ttlHours: number
  fetchFn?: typeof fetch
}): Promise<PricingTable> {
  const fetchFn = opts.fetchFn ?? fetch
  const cache = readCache(opts.cacheDir)
  const fresh = cache && Date.now() - cache.fetchedAt < opts.ttlHours * 3_600_000
  if (fresh) return cache.table
  try {
    const res = await fetchFn(PRICING_URL)
    if (!res.ok) throw new Error(`pricing fetch ${res.status}`)
    const table = (await res.json()) as PricingTable
    try {
      mkdirSync(opts.cacheDir, { recursive: true })
      writeFileSync(join(opts.cacheDir, 'pricing.json'), JSON.stringify({ fetchedAt: Date.now(), table }))
    } catch {}
    return table
  } catch {
    if (cache) return cache.table // stale beats bundled
    return readSnapshot()
  }
}

export function computeCost(table: PricingTable, provider: string, model: string, usage: Usage): number | null {
  const entry = table[model] ?? table[`${provider}/${model}`]
  if (!entry) return null
  const cached = usage.tokensCached ?? 0
  const uncachedIn = Math.max(0, (usage.tokensIn ?? 0) - cached)
  const cost =
    uncachedIn * (entry.input_cost_per_token ?? 0) +
    cached * (entry.cache_read_input_token_cost ?? entry.input_cost_per_token ?? 0) +
    (usage.tokensOut ?? 0) * (entry.output_cost_per_token ?? 0)
  return cost
}
```

```ts
import { existsSync, readFileSync } from 'node:fs'
import { basename, dirname, join } from 'node:path'

export function resolveProject(cwd: string, override?: string): { name: string; rootPath: string; gitRoot: string | null } {
  const gitRoot = findUp(cwd, (dir) => existsSync(join(dir, '.git')))
  const pkgDir = findUp(cwd, (dir) => existsSync(join(dir, 'package.json')))
  if (override) return { name: override, rootPath: pkgDir ?? cwd, gitRoot }
  if (pkgDir) {
    try {
      const name = JSON.parse(readFileSync(join(pkgDir, 'package.json'), 'utf8')).name
      if (typeof name === 'string' && name.length > 0) return { name, rootPath: pkgDir, gitRoot }
    } catch {}
    return { name: basename(pkgDir), rootPath: pkgDir, gitRoot }
  }
  if (gitRoot) return { name: basename(gitRoot), rootPath: gitRoot, gitRoot }
  return { name: basename(cwd), rootPath: cwd, gitRoot: null }
}

function findUp(from: string, pred: (dir: string) => boolean): string | null {
  let dir = from
  while (true) {
    if (pred(dir)) return dir
    const parent = dirname(dir)
    if (parent === dir) return null
    dir = parent
  }
}
```

```ts
import { existsSync, readFileSync } from 'node:fs'
import { homedir } from 'node:os'
import { dirname, join } from 'node:path'

export interface Burn0Config {
  project: string | null
  retentionDays: { payloads: number; metadata: number }
  bodyCapBytes: number
  ignoreHosts: string[]
  capturePayloads: boolean
  redact: { extraHeaders: string[]; disable: boolean }
  pricingTtlHours: number
  dbPath: string
  disabled: boolean
}

const CEILINGS = { payloads: 14, metadata: 90 }

export function loadConfig(cwd: string, env: NodeJS.ProcessEnv = process.env): Burn0Config {
  let raw: any = {}
  const file = findConfigFile(cwd)
  if (file) {
    try { raw = JSON.parse(readFileSync(file, 'utf8')) } catch {
      try { process.stderr.write('[burn0] invalid burn0.config.json ignored\n') } catch {}
    }
  }
  const num = (v: unknown, dflt: number, max?: number) =>
    typeof v === 'number' && v > 0 ? Math.min(v, max ?? Infinity) : dflt
  const disabled =
    env.BURN0_DISABLED === '1' ||
    (env.NODE_ENV === 'production' && env.BURN0_FORCE !== '1')
  return {
    project: env.BURN0_PROJECT ?? (typeof raw.project === 'string' ? raw.project : null),
    retentionDays: {
      payloads: num(raw.retentionDays?.payloads, 7, CEILINGS.payloads),
      metadata: num(raw.retentionDays?.metadata, 30, CEILINGS.metadata),
    },
    bodyCapBytes: num(raw.bodyCapBytes, 1048576),
    ignoreHosts: Array.isArray(raw.ignoreHosts) ? raw.ignoreHosts.filter((h: unknown) => typeof h === 'string') : [],
    capturePayloads: raw.capturePayloads !== false,
    redact: {
      extraHeaders: Array.isArray(raw.redact?.extraHeaders) ? raw.redact.extraHeaders : [],
      disable: raw.redact?.disable === true,
    },
    pricingTtlHours: num(raw.pricingTtlHours, 24),
    dbPath: env.BURN0_DB_PATH ?? join(homedir(), '.burn0', 'burn0.db'),
    disabled,
  }
}

function findConfigFile(from: string): string | null {
  let dir = from
  while (true) {
    const p = join(dir, 'burn0.config.json')
    if (existsSync(p)) return p
    const parent = dirname(dir)
    if (parent === dir) return null
    dir = parent
  }
}
```

```ts
import Database from 'better-sqlite3'

export interface PreparedStmt {
  run(...args: unknown[]): unknown
  get(...args: unknown[]): any
  all(...args: unknown[]): any[]
}

export interface StorageDriver {
  exec(sql: string): void
  prepare(sql: string): PreparedStmt
  transaction<A extends any[]>(fn: (...a: A) => void): (...a: A) => void
  pragma(s: string): unknown
  close(): void
}

// Thin adapter: better-sqlite3 today; bun:sqlite / node:sqlite later.
export function openBetterSqlite(path: string): StorageDriver {
  const db = new Database(path)
  return {
    exec: (sql) => db.exec(sql),
    prepare: (sql) => db.prepare(sql) as unknown as PreparedStmt,
    transaction: (fn) => db.transaction(fn) as any,
    pragma: (s) => db.pragma(s),
    close: () => db.close(),
  }
}
```

```ts
import { mkdirSync, renameSync, existsSync, rmSync } from 'node:fs'
import { dirname } from 'node:path'
import { SCHEMA_VERSION, type EnrichedRow } from '../types.js'
import { openBetterSqlite, type StorageDriver } from './driver.js'
import { DDL } from './schema.js'

export type { StorageDriver, PreparedStmt } from './driver.js'

export function openDatabase(dbPath: string): StorageDriver {
  mkdirSync(dirname(dbPath), { recursive: true })
  let db = openBetterSqlite(dbPath)
  db.pragma('journal_mode = WAL')
  db.exec(DDL)
  const row = db.prepare(`SELECT value FROM meta WHERE key='schema_version'`).get()
  if (!row) {
    db.prepare(`INSERT INTO meta (key, value) VALUES ('schema_version', ?)`).run(String(SCHEMA_VERSION))
  } else if (row.value !== String(SCHEMA_VERSION)) {
    db.close()
    const bak = dbPath + '.bak'
    if (existsSync(bak)) rmSync(bak)
    renameSync(dbPath, bak)
    for (const suffix of ['-wal', '-shm']) {
      if (existsSync(dbPath + suffix)) rmSync(dbPath + suffix)
    }
    db = openBetterSqlite(dbPath)
    db.pragma('journal_mode = WAL')
    db.exec(DDL)
    db.prepare(`INSERT INTO meta (key, value) VALUES ('schema_version', ?)`).run(String(SCHEMA_VERSION))
  }
  return db
}

export function ensureProject(db: StorageDriver, p: { name: string; rootPath: string; gitRoot: string | null }): number {
  db.prepare(`INSERT INTO projects (name, root_path, git_root, first_seen) VALUES (?, ?, ?, ?)
              ON CONFLICT(name) DO NOTHING`).run(p.name, p.rootPath, p.gitRoot, Date.now())
  return db.prepare(`SELECT id FROM projects WHERE name = ?`).get(p.name).id as number
}

export function insertRows(db: StorageDriver, projectId: number, rows: EnrichedRow[], capturePayloads: boolean): void {
  const insReq = db.prepare(`INSERT INTO requests
    (project_id, entity, entity_class, provider, method, url, normalized_endpoint, hostname,
     status, started_at, first_byte_ms, latency_ms, req_bytes, res_bytes,
     model, tokens_in, tokens_out, tokens_cached, cost_usd, estimated, tags)
    VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`)
  const insPay = db.prepare(`INSERT INTO payloads
    (request_id, req_headers, req_body, res_headers, res_body, stack_snippet, req_truncated, res_truncated)
    VALUES (?,?,?,?,?,?,?,?)`)
  const tx = db.transaction((batch: EnrichedRow[]) => {
    for (const r of batch) {
      const res: any = insReq.run(
        projectId, r.entity, r.entityClass, r.provider, r.method, r.url, r.normalizedEndpoint, r.hostname,
        r.status, r.startedAt, r.firstByteMs, r.latencyMs, r.reqBytes, r.resBytes,
        r.model, r.tokensIn, r.tokensOut, r.tokensCached, r.costUsd, r.estimated ? 1 : 0,
        r.tags ? JSON.stringify(r.tags) : null,

      if (capturePayloads) {

          JSON.stringify(r.resHeaders), r.resBody, r.stackSnippet, r.reqTruncated ? 1 : 0, r.resTruncated ? 1 : 0)

    }
```

```ts
import type { EnrichedRow } from '../types.js'

export class WriteQueue {
  private pending: EnrichedRow[] = []
  private timer: NodeJS.Timeout | null = null
  dropped = 0

  constructor(private opts: {
    flush: (rows: EnrichedRow[]) => void
    maxPending?: number
    intervalMs?: number
    maxBatch?: number
  }) {}

  push(row: EnrichedRow): void {
    try {
      this.pending.push(row)
      const maxPending = this.opts.maxPending ?? 5000
      while (this.pending.length > maxPending) {
        this.pending.shift()
        this.dropped++
      }
      if (this.pending.length >= (this.opts.maxBatch ?? 100)) {
        this.flushNow()
      } else if (!this.timer) {
        this.timer = setTimeout(() => this.flushNow(), this.opts.intervalMs ?? 250)
        this.timer.unref?.()
      }
    } catch {
      this.dropped++
    }
  }

  flushNow(): void {
    if (this.timer) { clearTimeout(this.timer); this.timer = null }
    if (this.pending.length === 0) return
    const batch = this.pending
    this.pending = []
    try {
      this.opts.flush(batch)
    } catch {
      this.dropped += batch.length
    }
  }

  stop(): void {
    this.flushNow()
    if (this.timer) { clearTimeout(this.timer); this.timer = null }
  }
}
```

```ts
  db.prepare(`DELETE FROM payloads WHERE request_id IN (SELECT id FROM requests WHERE started_at < ?)`)

  const row = db.prepare(`SELECT value FROM meta WHERE key='last_vacuum'`).get()

  db.exec('VACUUM')

              ON CONFLICT(key) DO UPDATE SET value = excluded.value`).run(String(now))
```

```ts
import { detectEntity, normalizeEndpoint } from '../entities/index.js'
import { redactHeaders, redactBody } from '../storage/redact.js'
import { extractModel, parseUsage, parseSSEUsage } from '../llm/parse.js'
import { computeCost, type PricingTable } from '../pricing/index.js'
import type { CaptureEvent, EnrichedRow } from '../types.js'

export interface EnrichOpts {
  pricing: PricingTable
  bodyCapBytes: number
  redactDisabled: boolean
  extraRedactHeaders: string[]
  isSSE: boolean
}

export function enrich(event: CaptureEvent, opts: EnrichOpts): EnrichedRow {
  const { entity, entityClass, provider } = detectEntity(event.hostname)
  let pathname = '/'
  try { pathname = new URL(event.url).pathname } catch {}

  let model: string | null = null
  let tokensIn: number | null = null
  let tokensOut: number | null = null
  let tokensCached: number | null = null
  let costUsd: number | null = null
  if (entityClass === 'llm') {
    model = extractModel(event.reqBody, pathname, opts.isSSE ? null : event.resBody)
    const usage = opts.isSSE ? parseSSEUsage(event.resBody ?? '') : parseUsage(event.resBody)
    if (usage) {
      tokensIn = usage.tokensIn
      tokensOut = usage.tokensOut
      tokensCached = usage.tokensCached
      if (model && provider) costUsd = computeCost(opts.pricing, provider, model, usage)
    }
  }

  const [reqBody, reqTruncated] = cap(event.reqBody, opts.bodyCapBytes)
  const [resBody, resTruncated] = cap(event.resBody, opts.bodyCapBytes)

  return {
    ...event,
    reqHeaders: opts.redactDisabled ? event.reqHeaders : redactHeaders(event.reqHeaders, opts.extraRedactHeaders),
    resHeaders: opts.redactDisabled ? event.resHeaders : redactHeaders(event.resHeaders, opts.extraRedactHeaders),
    reqBody: opts.redactDisabled ? reqBody : redactBody(reqBody),
    resBody: opts.redactDisabled ? resBody : redactBody(resBody),
    entity, entityClass, provider,
    normalizedEndpoint: normalizeEndpoint(pathname),
    model, tokensIn, tokensOut, tokensCached, costUsd,
    reqTruncated, resTruncated,
  }
}

function cap(body: string | null, max: number): [string | null, boolean] {
  if (body == null || body.length <= max) return [body, false]
  return [body.slice(0, max), true]
}
```

```ts
  const timeout = new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error('stream timeout')), timeoutMs)
    timer.unref?.()
  })
  try {
    while (true) {
      const result = (await Promise.race([reader.read(), timeout])) as ReadableStreamReadResult<Uint8Array>
      if (result.done) break
      bytes += result.value.byteLength
      if (bytes <= maxBytes + 65536) chunks.push(decoder.decode(result.value, { stream: true }))
      // keep draining past cap so tee's internal queue frees, but stop storing
    }
  } catch {
    // timeout / stream error: return what we have (partial)
  } finally {
    if (timer) clearTimeout(timer)
    try { reader.releaseLock() } catch {}
  }
  return chunks.join('')
}
```

```ts
export function hostIgnored(hostname: string, patterns: string[]): boolean {
  return patterns.some((p) =>
    p.startsWith('*.') ? hostname.endsWith(p.slice(1)) || hostname === p.slice(2) : hostname === p)
}

function headersToRecord(h: Headers | Record<string, any> | string[][] | undefined): Record<string, string> {
  const out: Record<string, string> = {}
  try {
    if (!h) return out
    if (h instanceof Headers) { h.forEach((v, k) => { out[k] = v }); return out }
    if (Array.isArray(h)) { for (const [k, v] of h) out[k] = String(v); return out }
    for (const [k, v] of Object.entries(h)) out[k] = String(v)
  } catch {}
  return out
}

export function patchFetch(emit: EmitFn, opts: CaptureOpts): () => void {
  const original = globalThis.fetch
  const safeEmit: EmitFn = (e, sse) => { try { emit(e, sse) } catch {} }

  globalThis.fetch = async function burn0Fetch(input: RequestInfo | URL, init?: RequestInit): Promise<Response> {
    // Anything that throws in our pre/post logic must not affect the app call.
    let url: URL | null = null
    try {
      url = new URL(typeof input === 'string' ? input : input instanceof URL ? input.href : input.url)
    } catch {}
    if (!url || hostIgnored(url.hostname, opts.ignoreHosts)) {
      return runMarked(() => original(input, init))
    }

    const startedAt = Date.now()
    const stackSnippet = captureStackSnippet()
    const tags = currentTags()
    const method = (init?.method ?? (typeof input === 'object' && 'method' in input ? input.method : 'GET')).toUpperCase()
    const reqHeaders = headersToRecord(init?.headers as any ?? (typeof input === 'object' && 'headers' in input ? (input as Request).headers : undefined))
    const reqBody = typeof init?.body === 'string' ? init.body : null

    let response: Response
    try {
      response = await runMarked(() => original(input, init))
    } catch (err) {
      // network failure: record it, then rethrow untouched
      safeEmit({
        startedAt, method, url: url.href, hostname: url.hostname, status: null,
        firstByteMs: null, latencyMs: Date.now() - startedAt,
        reqBytes: reqBody?.length ?? 0, resBytes: 0,
        reqHeaders, resHeaders: {}, reqBody, resBody: null,
        stackSnippet, tags, estimated: false,
      }, false)
      throw err
    }

    try {
      const firstByteMs = Date.now() - startedAt
      const resHeaders = headersToRecord(response.headers)
      const base = {
        startedAt, method, url: url.href, hostname: url.hostname, status: response.status,
        firstByteMs, reqBytes: reqBody?.length ?? 0,
        reqHeaders, resHeaders, reqBody, stackSnippet, tags, estimated: false,
      }
      const contentType = response.headers.get('content-type') ?? ''

      if (contentType.includes('text/event-stream') && response.body) {
        const [forApp, forBurn0] = response.body.tee()
        collectStream(forBurn0, opts.bodyCapBytes).then((raw) => {
          safeEmit({ ...base, latencyMs: Date.now() - startedAt, resBytes: raw.length, resBody: raw, estimated: raw.length === 0 }, true)
        }).catch(() => {
          safeEmit({ ...base, latencyMs: Date.now() - startedAt, resBytes: 0, resBody: null, estimated: true }, true)
        })
        return new Response(forApp, { status: response.status, statusText: response.statusText, headers: response.headers })
      }

      // Non-stream: clone and read async; app gets original response immediately
      response.clone().text().then((text) => {
        safeEmit({ ...base, latencyMs: Date.now() - startedAt, resBytes: text.length, resBody: text }, false)
      }).catch(() => {
        safeEmit({ ...base, latencyMs: Date.now() - startedAt, resBytes: 0, resBody: null }, false)
      })
      return response
    } catch {
      return response  // burn0 post-processing failed: app still gets its response
    }
  }

  return () => { globalThis.fetch = original }
}
```

```ts
import http from 'node:http'
import https from 'node:https'
import { captureStackSnippet } from './stack.js'
import { currentTags } from '../context/index.js'
import { hostIgnored, type EmitFn, type CaptureOpts } from './fetch-patch.js'

type RequestFn = typeof http.request

export function patchHttp(emit: EmitFn, opts: CaptureOpts): () => void {
  const originals = {
    httpRequest: http.request, httpGet: http.get,
    httpsRequest: https.request, httpsGet: https.get,
  }
  const safeEmit: EmitFn = (e, sse) => { try { emit(e, sse) } catch {} }

  const wrap = (original: RequestFn, protocol: 'http:' | 'https:'): RequestFn =>
    function burn0Request(this: unknown, ...args: any[]) {
      const req = (original as any).apply(this, args)
      try {
        instrument(req, args, protocol, safeEmit, opts)
      } catch {}
      return req
    } as RequestFn

  http.request = wrap(originals.httpRequest, 'http:')
  https.request = wrap(originals.httpsRequest, 'https:')
  http.get = ((...args: any[]) => { const r = (http.request as any)(...args); r.end(); return r }) as typeof http.get
  https.get = ((...args: any[]) => { const r = (https.request as any)(...args); r.end(); return r }) as typeof https.get

  return () => {
    http.request = originals.httpRequest
    http.get = originals.httpGet
    https.request = originals.httpsRequest
    https.get = originals.httpsGet
  }
}

function instrument(req: http.ClientRequest, args: any[], protocol: string, emit: EmitFn, opts: CaptureOpts): void {
  const startedAt = Date.now()
  const stackSnippet = captureStackSnippet()
  const tags = currentTags()

  // Reconstruct URL from (url[, options]) or (options) call signatures
  let urlStr: string
  const a0 = args[0]
  if (typeof a0 === 'string') urlStr = a0
  else if (a0 instanceof URL) urlStr = a0.href
  else {
    const o = a0 ?? {}
    const host = o.hostname ?? o.host ?? 'localhost'
    const port = o.port ? `:${o.port}` : ''
    urlStr = `${o.protocol ?? protocol}//${host}${port}${o.path ?? '/'}`
  }
  let url: URL
  try { url = new URL(urlStr) } catch { return }
  if (hostIgnored(url.hostname, opts.ignoreHosts)) return

  // capture request body via write/end interception
  const reqChunks: string[] = []
  let reqBytes = 0
  const origWrite = req.write.bind(req)
  const origEnd = req.end.bind(req)
  req.write = ((chunk: any, ...rest: any[]) => {
    try {
      if (chunk != null) {
        reqBytes += Buffer.byteLength(chunk)
        if (reqBytes <= opts.bodyCapBytes + 65536) reqChunks.push(String(chunk))
      }
    } catch {}
    return origWrite(chunk, ...rest)
  }) as any
  req.end = ((chunk?: any, ...rest: any[]) => {
    try {
      if (chunk != null && typeof chunk !== 'function') {
        reqBytes += Buffer.byteLength(chunk)
        if (reqBytes <= opts.bodyCapBytes + 65536) reqChunks.push(String(chunk))
      }
    } catch {}
    return origEnd(chunk, ...rest)
  }) as any

  req.on('response', (res: http.IncomingMessage) => {
    try {
      const firstByteMs = Date.now() - startedAt
      const resChunks: string[] = []
      let resBytes = 0
      res.on('data', (chunk: Buffer) => {
        try {
          resBytes += chunk.length
          if (resBytes <= opts.bodyCapBytes + 65536) resChunks.push(chunk.toString('utf8'))
        } catch {}
      })
      res.on('end', () => {
        const contentType = String(res.headers['content-type'] ?? '')
        const isSSE = contentType.includes('text/event-stream')
        emit({
          startedAt, method: (req.method ?? 'GET').toUpperCase(), url: url.href, hostname: url.hostname,
          status: res.statusCode ?? null, firstByteMs, latencyMs: Date.now() - startedAt,
          reqBytes, resBytes,
          reqHeaders: flattenHeaders(req.getHeaders()), resHeaders: flattenHeaders(res.headers),
          reqBody: reqChunks.length ? reqChunks.join('') : null,
          resBody: resChunks.length ? resChunks.join('') : null,
          stackSnippet, tags, estimated: false,
        }, isSSE)
      })
    } catch {}
  })

  req.on('error', () => {
    try {
      emit({
        startedAt, method: (req.method ?? 'GET').toUpperCase(), url: url.href, hostname: url.hostname,
        status: null, firstByteMs: null, latencyMs: Date.now() - startedAt,
        reqBytes, resBytes: 0,
        reqHeaders: flattenHeaders(req.getHeaders()), resHeaders: {},
        reqBody: reqChunks.length ? reqChunks.join('') : null, resBody: null,
        stackSnippet, tags, estimated: false,
      }, false)
    } catch {}
  })
}

function flattenHeaders(h: Record<string, unknown>): Record<string, string> {
  const out: Record<string, string> = {}
  try {
    for (const [k, v] of Object.entries(h)) {
      if (v == null) continue
      out[k] = Array.isArray(v) ? v.join(', ') : String(v)
    }
  } catch {}
  return out
}
```

Note: `res.on('data', ...)` adds a listener alongside the app's — it does NOT consume the stream exclusively; Node streams multicast `data` events to all listeners. If the app never reads the response, our listener switches the stream to flowing mode; that matches DevTools-style capture semantics and is acceptable for a dev tool.

```ts
import { subscribe, unsubscribe } from 'node:diagnostics_channel'
import { captureStackSnippet } from './stack.js'
import { currentTags } from '../context/index.js'
import { isMarked } from './dedup.js'
import { hostIgnored, type EmitFn, type CaptureOpts } from './fetch-patch.js'

interface Pending {
  startedAt: number
  method: string
  url: URL
  reqHeaders: Record<string, string>
  stackSnippet: string | null
  tags: Record<string, string> | null
  status: number | null
  resHeaders: Record<string, string>
  firstByteMs: number | null
  skip: boolean
}

export function subscribeUndici(emit: EmitFn, opts: CaptureOpts): () => void {
  const pending = new WeakMap<object, Pending>()
  const safeEmit: EmitFn = (e, sse) => { try { emit(e, sse) } catch {} }

  const onCreate = (message: any) => {
    try {
      const req = message.request
      if (!req) return
      const url = new URL(req.origin ? `${req.origin}${req.path ?? '/'}` : req.path)
      pending.set(req, {
        startedAt: Date.now(),
        method: (req.method ?? 'GET').toUpperCase(),
        url,
        reqHeaders: parseRawHeaders(req.headers),
        stackSnippet: captureStackSnippet(),
        tags: currentTags(),
        status: null, resHeaders: {}, firstByteMs: null,
        skip: isMarked() || hostIgnored(url.hostname, opts.ignoreHosts),
      })
    } catch {}
  }

  const onHeaders = (message: any) => {
    try {
      const p = pending.get(message.request)
      if (!p) return
      p.firstByteMs = Date.now() - p.startedAt
      p.status = message.response?.statusCode ?? null
      p.resHeaders = parseRawHeaderPairs(message.response?.headers)
    } catch {}
  }

  const onDone = (message: any) => {
    try {
      const p = pending.get(message.request)
      if (!p || p.skip) return
      pending.delete(message.request)
      safeEmit({
        startedAt: p.startedAt, method: p.method, url: p.url.href, hostname: p.url.hostname,
        status: p.status, firstByteMs: p.firstByteMs, latencyMs: Date.now() - p.startedAt,
        reqBytes: 0, resBytes: 0,
        reqHeaders: p.reqHeaders, resHeaders: p.resHeaders,
        reqBody: null, resBody: null,
        stackSnippet: p.stackSnippet, tags: p.tags, estimated: false,
      }, false)
    } catch {}
  }

  subscribe('undici:request:create', onCreate)
  subscribe('undici:request:headers', onHeaders)
  subscribe('undici:request:trailers', onDone)

  return () => {
    unsubscribe('undici:request:create', onCreate)
    unsubscribe('undici:request:headers', onHeaders)
    unsubscribe('undici:request:trailers', onDone)
  }
}

function parseRawHeaders(h: unknown): Record<string, string> {
  const out: Record<string, string> = {}
  try {
    if (typeof h === 'string') {
      for (const line of h.split('\r\n')) {
        const i = line.indexOf(':')
        if (i > 0) out[line.slice(0, i).trim().toLowerCase()] = line.slice(i + 1).trim()
      }
    } else if (Array.isArray(h)) {
      for (let i = 0; i + 1 < h.length; i += 2) out[String(h[i]).toLowerCase()] = String(h[i + 1])
    } else if (h && typeof h === 'object') {
      for (const [k, v] of Object.entries(h)) out[k.toLowerCase()] = String(v)
    }
  } catch {}
  return out
}

function parseRawHeaderPairs(h: unknown): Record<string, string> {
  const out: Record<string, string> = {}
  try {
    if (Array.isArray(h)) {
      for (let i = 0; i + 1 < h.length; i += 2) out[String(h[i]).toLowerCase()] = String(h[i + 1])
    }
  } catch {}
  return out
}
```

```ts
export function overview(db: StorageDriver, now = Date.now()) {
```

```ts
import { dirname } from 'node:path'
import { loadConfig } from './config/index.js'
import { resolveProject } from './project/index.js'
import { openDatabase, ensureProject, insertRows, type StorageDriver } from './storage/db.js'
import { WriteQueue } from './storage/queue.js'
import { pruneRetention, maybeVacuum } from './storage/retention.js'
import { loadPricing, type PricingTable } from './pricing/index.js'
import { enrich } from './capture/enrich.js'
import { patchFetch, type EmitFn } from './capture/fetch-patch.js'
import { patchHttp } from './capture/http-patch.js'
import { subscribeUndici } from './capture/undici-dc.js'
import type { CaptureEvent } from './types.js'

export interface Burn0Handle {
  queue: WriteQueue
  db: StorageDriver
  counters: { errors: number; dropped: () => number }
  shutdown(): void
}

let active: Burn0Handle | null = null

export function initBurn0(overrides: { cwd?: string; env?: NodeJS.ProcessEnv } = {}): Burn0Handle | null {
  if (active) return active
  try {
    // Runtime gate: Node only (dev tool; bun native support is roadmap)
    const isNode = typeof process !== 'undefined' && !!process.versions?.node &&
      !(process.versions as any).bun && !(globalThis as any).Deno
    if (!isNode) {
      try { process.stderr.write('[burn0] unsupported runtime, capture disabled\n') } catch {}
      return null
    }

    const cwd = overrides.cwd ?? process.cwd()
    const cfg = loadConfig(cwd, overrides.env ?? process.env)
    if (cfg.disabled) return null

    const project = resolveProject(cwd, cfg.project ?? undefined)
    const db = openDatabase(cfg.dbPath)
    const projectId = ensureProject(db, project)
    try { pruneRetention(db, cfg.retentionDays); maybeVacuum(db) } catch {}

    const counters = { errors: 0, dropped: () => queue.dropped }

    // Pricing: start empty, refresh async (fetch → cache → bundled snapshot)
    let pricing: PricingTable = {}
    loadPricing({ cacheDir: dirname(cfg.dbPath), ttlHours: cfg.pricingTtlHours })
      .then((t) => { pricing = t })
      .catch(() => { counters.errors++ })

    const queue = new WriteQueue({
      flush: (rows) => insertRows(db, projectId, rows, cfg.capturePayloads),
    })

    const emit: EmitFn = (event: CaptureEvent, isSSE: boolean) => {
      try {
        queue.push(enrich(event, {
          pricing, bodyCapBytes: cfg.bodyCapBytes,
          redactDisabled: cfg.redact.disable, extraRedactHeaders: cfg.redact.extraHeaders,
          isSSE,
        }))
      } catch { counters.errors++ }
    }

    const captureOpts = { ignoreHosts: cfg.ignoreHosts, bodyCapBytes: cfg.bodyCapBytes }
    const unpatchFetch = patchFetch(emit, captureOpts)
    const unpatchHttp = patchHttp(emit, captureOpts)
    const unsubDc = subscribeUndici(emit, captureOpts)

    const flushOnExit = () => { try { queue.flushNow() } catch {} }
    process.on('beforeExit', flushOnExit)
    process.on('exit', flushOnExit)

    active = {
      queue, db, counters,
      shutdown() {
        try { unpatchFetch(); unpatchHttp(); unsubDc(); queue.stop(); db.close() } catch {}
        process.removeListener('beforeExit', flushOnExit)
        process.removeListener('exit', flushOnExit)
        active = null
      },
    }
    return active
  } catch {
    // Global Constraint: init failure must never crash the host app
    return null
  }
}
```
